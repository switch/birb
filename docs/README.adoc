= Birb - quick start guide

== Installation

Birb uses https://github.com/vektra/mockery[mockery] to generate mocks.

https://vektra.github.io/mockery/latest/installation/[install mockery]

Then install Birb:
[source,console]
$ go install github.com/switch/birb/cmd/birb@latest

== Configuration

.run birb
[source,console]
----
& birb
Birb Mockery Template Generator

Generating mockery template directory...
Generating mockery template...
Generating mockery template schema...
----

..mockery.yaml
[,yaml]
----
template: file://templates/mockery.go.tmpl
packages:
  github.com/switch/birb/test/e2e:
    interfaces:
      Greeter:
----

See the https://vektra.github.io/mockery/latest/configuration/[mockery documentation] for more information on the configuration file.

The Greeter Interface:
[,go]
----
package e2e

type Greeter interface {
	Greet() string
	PersonalGreet(name string) (string, error)
	AllTheGreets(who []string) []string
}
----

== Usage

=== Generate the files you want to mock with Birb

.run mockery
[source,console]
& mockery

=== Use the mocks in your tests

==== Ginkgo example
[,go]
----
package e2e

import (
	. "github.com/switch/birb"
	. "github.com/onsi/ginkgo/v2"
	. "github.com/onsi/gomega"
)

var _ = Describe("ginkgo integration - Greeter", func() {
	var mockGreeter *MockGreeter // MockGreeter is the mock generated by Birb for the Greeter interface

	BeforeEach(func() {
		mockGreeter = NewMockGreeter(GinkgoT())
	})

	AfterEach(func() {
		VerifyNoOtherInteractions(mockGreeter)
	})

	Describe("Greet", func() {
		It("should verify a mock hasn't been touched", func() {
			VerifyNeverCalled(mockGreeter)
			Freeze(mockGreeter)
		})

		It("should run a simple test", func() {
			WhenCalling(mockGreeter.MOCK_Greet()).ThenReturn("Hello, John")

			actual := mockGreeter.Greet()

			Expect(actual).To(Equal("Hello, John"))

			Verify(mockGreeter, Once()).CALLED_Greet()
		})
	})
})

----

==== Go Test example
[,go]
----
package e2e

import (
	"testing"

	. "github.com/switch/birb"
)

func TestGreeter(t *testing.T) {
	mockGreeter := NewMockGreeter(t) // MockGreeter is the mock generated by Birb for the Greeter interface
	WhenCalling(mockGreeter.MOCK_Greet()).ThenReturn("Hello, John")

	actual := mockGreeter.Greet()

	if actual != "Hello, John" {
		t.Errorf("Expected 'Hello, John', got '%s'", actual)
	}
	Verify(mockGreeter, Once()).CALLED_Greet()
	VerifyNoOtherInteractions(mockGreeter)
}

----
